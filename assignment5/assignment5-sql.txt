# Task 1: the total price of the first 5 orders.
SELECT
  orders.order_id,
  SUM(products.price * line_items.quantity) AS total_price
FROM orders
JOIN line_items ON orders.order_id = line_items.order_id
JOIN products ON line_items.product_id = products.product_id
GROUP BY orders.order_id
ORDER BY orders.order_id
LIMIT 5;

# Task 2: the average order total for each customer. The subquery returns the
# customer_id and the total price of every order, and the outer query averages
# those totals for each customer.
SELECT
  customer_name,
  AVG(t.total_price) AS average_order_price
FROM customers c
JOIN (
  SELECT
    orders.customer_id,
    SUM(products.price * line_items.quantity) AS total_price
  FROM orders
  JOIN line_items ON orders.order_id = line_items.order_id
  JOIN products ON line_items.product_id = products.product_id
  GROUP BY orders.order_id, orders.customer_id
) AS t ON c.customer_id = t.customer_id
GROUP BY c.customer_id, customer_name
ORDER BY customer_name;

# Task 4: the employees who have taken more than 5 orders.
SELECT
  first_name,
  last_name,
  COUNT(order_id) AS order_count
FROM employees
JOIN orders ON employees.employee_id = orders.employee_id
GROUP BY employees.employee_id, first_name, last_name
HAVING COUNT(order_id) > 5
ORDER BY last_name;
